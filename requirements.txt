# pip dep below resolves the per-user data dir on Mac/Windows/Linux when the
# configured storage_path is unavailable (e.g. Pi USB not mounted, or a fresh
# install on a non-Pi platform).

pyobjc-framework-Cocoa>=10.0; sys_platform == "darwin"
